    const GOOGLE_SCRIPT_URL = "https://script.google.com/macros/s/AKfycby9VUoJOr2g-kFlT03PsXrNiJGlzywiw3sZ8xvMPf7oNpDunSn460skGut4KUZVgOrSnA/exec";

    

    const CONFIG = {

      TOTAL_TIME: 30,

      POINTS_PER_HIT: 10,

      PENALTY_PER_HIT: 10,

      NORMAL_SPEED_MS: 800,

      FRENZY_SPEED_MS: 400

    };


    let score = 0;

    let timeLeft = CONFIG.TOTAL_TIME;

    let gameTimer = null;

    let moveTimer = null;

    let isPlaying = false;

    let currentSpeed = CONFIG.NORMAL_SPEED_MS;


    const crowBtn = document.getElementById('crowBtn');

    const trapBtn = document.getElementById('trapBtn');

    const arena = document.getElementById('gameArena');

    const frenzyBadge = document.getElementById('frenzyBadge');

    const timerText = document.getElementById('timerText');

    const scoreText = document.getElementById('scoreText');

    const topScoreText = document.getElementById('topScoreText');

    const startScreen = document.getElementById('startScreen');

    const gameOverScreen = document.getElementById('gameOverScreen');

    const finalScoreVal = document.getElementById('finalScoreVal');

    const playerNameInput = document.getElementById('playerNameInput');

    const saveScoreBtn = document.getElementById('saveScoreBtn');

    const saveBtnText = document.getElementById('saveBtnText');

    const saveStatusMsg = document.getElementById('saveStatusMsg');

    const playAgainBtn = document.getElementById('playAgainBtn');

    const startGameBtn = document.getElementById('startGameBtn');

    const leaderboardList = document.getElementById('leaderboardList');

    const refreshBoardBtn = document.getElementById('refreshBoardBtn');


    let audioCtx = null;

    function initAudio() {

      if (!audioCtx) {

        audioCtx = new (window.AudioContext || window.webkitAudioContext)();

      }

      if (audioCtx.state === 'suspended') {

        audioCtx.resume();

      }

    }


    function playShotTone() {

      try {

        initAudio();

        const osc = audioCtx.createOscillator();

        const gain = audioCtx.createGain();

        osc.type = 'sawtooth';

        osc.frequency.setValueAtTime(850, audioCtx.currentTime);

        osc.frequency.exponentialRampToValueAtTime(120, audioCtx.currentTime + 0.08);

        gain.gain.setValueAtTime(0.25, audioCtx.currentTime);

        gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.08);

        osc.connect(gain);

        gain.connect(audioCtx.destination);

        osc.start();

        osc.stop(audioCtx.currentTime + 0.08);

      } catch (e) {}

    }


    function playPenaltyTone() {

      try {

        initAudio();

        const osc = audioCtx.createOscillator();

        const gain = audioCtx.createGain();

        osc.type = 'square';

        osc.frequency.setValueAtTime(160, audioCtx.currentTime);

        osc.frequency.linearRampToValueAtTime(80, audioCtx.currentTime + 0.18);

        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);

        gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.18);

        osc.connect(gain);

        gain.connect(audioCtx.destination);

        osc.start();

        osc.stop(audioCtx.currentTime + 0.18);

      } catch (e) {}

    }


    async function fetchTop10Leaderboard() {

      leaderboardList.innerHTML = '<li class="text-center text-slate-500 italic py-2">Mengambil senarai 10 terbaik...</li>';


      try {

        const response = await fetch(GOOGLE_SCRIPT_URL, {

          method: 'GET',

          redirect: 'follow'

        });


        if (!response.ok) throw new Error('Respons rangkaian gagal');


        const data = await response.json();


        if (!data || !data.length) {

          leaderboardList.innerHTML = '<li class="text-center text-slate-500 italic py-2">Belum ada pemain dalam Google Sheet.</li>';

          topScoreText.textContent = '0';

          return;

        }


        topScoreText.textContent = data[0].score || 0;

        const badges = ['🥇', '🥈', '🥉'];


        leaderboardList.innerHTML = data.slice(0, 10).map((item, idx) => `

          <li class="flex items-center justify-between bg-slate-950/80 px-2.5 py-1.5 rounded-lg border border-slate-800">

            <div class="flex items-center gap-2 truncate pr-2">

              <span class="w-4 text-center font-bold text-[10px] ${idx < 3 ? '' : 'text-slate-500'}">

                ${badges[idx] || '#' + (idx + 1)}

              </span>

              <span class="text-slate-200 font-medium text-xs truncate">${escapeHtml(item.name)}</span>

            </div>

            <div class="flex items-center gap-2 shrink-0">

              <span class="text-[9px] text-slate-500 hidden sm:inline">${item.time || ''}</span>

              <span class="font-mono font-bold text-amber-400 bg-amber-400/10 px-1.5 py-0.5 rounded text-xs border border-amber-400/20">

                ${item.score} pts

              </span>

            </div>

          </li>

        `).join('');


      } catch (err) {

        console.error("Ralat Leaderboard:", err);

        leaderboardList.innerHTML = '<li class="text-center text-rose-400 py-2">Gagal berhubung dengan Google Sheet.</li>';

      }

    }


    async function submitScoreToSheet() {

      const rawName = playerNameInput.value.trim();

      const name = rawName.length > 0 ? rawName : 'Pemain Anon';


      saveScoreBtn.disabled = true;

      saveBtnText.textContent = "...";

      saveStatusMsg.textContent = "Menghantar ke Google Sheet Drive...";

      saveStatusMsg.className = "text-[10px] text-amber-400 font-semibold";


      try {

        await fetch(GOOGLE_SCRIPT_URL, {

          method: 'POST',

          mode: 'no-cors',

          headers: { 'Content-Type': 'text/plain;charset=utf-8' },

          body: JSON.stringify({ name: name, score: score })

        });


        saveStatusMsg.textContent = "Tersimpan dalam Google Sheet!";

        saveStatusMsg.className = "text-[10px] text-emerald-400 font-semibold";


        setTimeout(() => {

          gameOverScreen.classList.add('hidden');

          startScreen.classList.remove('hidden');

          saveScoreBtn.disabled = false;

          saveBtnText.textContent = "Simpan";

          saveStatusMsg.textContent = "";

          fetchTop10Leaderboard();

        }, 1100);


      } catch (err) {

        console.error("Ralat Simpan:", err);

        saveStatusMsg.textContent = "Ralat semasa menyimpan ke Sheet.";

        saveStatusMsg.className = "text-[10px] text-rose-400 font-semibold";

        saveScoreBtn.disabled = false;

        saveBtnText.textContent = "Simpan";

      }

    }


    function escapeHtml(str) {

      return String(str).replace(/[&<>"']/g, m => ({

        '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'

      }[m]));

    }


    function getRandomPosition() {

      const arenaW = arena.clientWidth;

      const arenaH = arena.clientHeight;

      const padding = 36;

      const maxX = arenaW - padding * 2;

      const maxY = arenaH - padding * 2;


      return {

        x: Math.floor(Math.random() * Math.max(maxX, 10)) + padding,

        y: Math.floor(Math.random() * Math.max(maxY, 10)) + padding

      };

    }


    function moveEntities() {

      if (!isPlaying) return;


      const crowPos = getRandomPosition();

      crowBtn.style.left = `${crowPos.x}px`;

      crowBtn.style.top = `${crowPos.y}px`;


      if (Math.random() < 0.65) {

        const trapPos = getRandomPosition();

        trapBtn.style.left = `${trapPos.x}px`;

        trapBtn.style.top = `${trapPos.y}px`;

        trapBtn.classList.remove('hidden');

      } else {

        trapBtn.classList.add('hidden');

      }

    }


    function resetMoveTimer(speed) {

      clearInterval(moveTimer);

      moveTimer = setInterval(moveEntities, speed);

    }


    function spawnHitPopup(x, y, text, isPenalty = false) {

      const popup = document.createElement('span');

      popup.className = `hit-popup absolute font-black font-mono text-base sm:text-lg drop-shadow z-20 ${

        isPenalty ? 'text-rose-500 scale-125' : 'text-emerald-400'

      }`;

      popup.textContent = text;

      popup.style.left = `${x}px`;

      popup.style.top = `${y}px`;

      arena.appendChild(popup);

      setTimeout(() => popup.remove(), 500);

    }


    function handleShot(e) {

      if (!isPlaying) return;

      e.preventDefault();

      e.stopPropagation();

      playShotTone();


      score += CONFIG.POINTS_PER_HIT;

      scoreText.textContent = score;


      const rect = arena.getBoundingClientRect();

      const clientX = e.clientX || (e.touches && e.touches[0].clientX) || 0;

      const clientY = e.clientY || (e.touches && e.touches[0].clientY) || 0;

      

      spawnHitPopup(clientX - rect.left, clientY - rect.top, `+${CONFIG.POINTS_PER_HIT}`);


      moveEntities();

      resetMoveTimer(currentSpeed);

    }


    function handleTrapShot(e) {

      if (!isPlaying) return;

      e.preventDefault();

      e.stopPropagation();

      playPenaltyTone();


      score = Math.max(0, score - CONFIG.PENALTY_PER_HIT);

      scoreText.textContent = score;


      arena.classList.add('shake-danger');

      setTimeout(() => arena.classList.remove('shake-danger'), 250);


      const rect = arena.getBoundingClientRect();

      const clientX = e.clientX || (e.touches && e.touches[0].clientX) || 0;

      const clientY = e.clientY || (e.touches && e.touches[0].clientY) || 0;


      spawnHitPopup(clientX - rect.left, clientY - rect.top, `-${CONFIG.PENALTY_PER_HIT} 💥`, true);


      trapBtn.classList.add('hidden');

      moveEntities();

      resetMoveTimer(currentSpeed);

    }


    function startGame() {

      initAudio();

      score = 0;

      timeLeft = CONFIG.TOTAL_TIME;

      currentSpeed = CONFIG.NORMAL_SPEED_MS;

      isPlaying = true;


      scoreText.textContent = '0';

      timerText.textContent = `${timeLeft}s`;

      timerText.className = "text-xl sm:text-2xl md:text-3xl font-black text-rose-400 font-mono transition-all";

      saveStatusMsg.textContent = '';

      frenzyBadge.classList.add('hidden');

      crowBtn.classList.remove('crow-fast');

      trapBtn.classList.remove('crow-fast');


      startScreen.classList.add('hidden');

      gameOverScreen.classList.add('hidden');

      crowBtn.classList.remove('hidden');


      moveEntities();

      resetMoveTimer(currentSpeed);


      gameTimer = setInterval(() => {

        timeLeft--;

        timerText.textContent = `${timeLeft}s`;


        if (timeLeft <= 10 && currentSpeed !== CONFIG.FRENZY_SPEED_MS) {

          currentSpeed = CONFIG.FRENZY_SPEED_MS;

          resetMoveTimer(currentSpeed);

          

          frenzyBadge.classList.remove('hidden');

          crowBtn.classList.add('crow-fast');

          trapBtn.classList.add('crow-fast');

          timerText.className = "text-2xl sm:text-3xl md:text-4xl font-black text-rose-500 font-mono animate-ping transition-all";

        }


        if (timeLeft <= 0) {

          endGame();

        }

      }, 1000);

    }


    function endGame() {

      isPlaying = false;

      clearInterval(gameTimer);

      clearInterval(moveTimer);


      frenzyBadge.classList.add('hidden');

      crowBtn.classList.remove('crow-fast');

      trapBtn.classList.remove('crow-fast');

      crowBtn.classList.add('hidden');

      trapBtn.classList.add('hidden');


      finalScoreVal.textContent = score;

      gameOverScreen.classList.remove('hidden');

      playerNameInput.value = '';

      setTimeout(() => playerNameInput.focus(), 200);

    }


    crowBtn.addEventListener('pointerdown', handleShot);

    trapBtn.addEventListener('pointerdown', handleTrapShot);

    startGameBtn.addEventListener('click', startGame);

    playAgainBtn.addEventListener('click', startGame);

    saveScoreBtn.addEventListener('click', submitScoreToSheet);

    refreshBoardBtn.addEventListener('click', fetchTop10Leaderboard);


    playerNameInput.addEventListener('keydown', (e) => {

      if (e.key === 'Enter') submitScoreToSheet();

    });


    arena.addEventListener('pointerdown', (e) => {

      if (e.target === arena) initAudio();

    });


    fetchTop10Leaderboard();
